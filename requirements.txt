# Core (needed everywhere)

# Windows clipboard (read is via Pillow; write needs pywin32)
pywin32>=306 ; sys_platform == "win32"

# OCR. RapidOCR (ONNX, CPU) is the DEFAULT — reliable across Python versions and
# bundles cleanly into the .exe (ship the models). winocr (built-in Windows
# engine) is lighter but depends on winrt, which can fail to load on some
# Python/Windows combos — kept as an optional faster path, auto-falls-back to
# RapidOCR. pytesseract needs the Tesseract binary installed separately.
rapidocr-onnxruntime>=1.3        # default neural OCR (CPU, ~15MB models)
winocr>=0.0.13 ; sys_platform == "win32"   # optional faster path (may need winrt)
# pytesseract>=0.3.10            # optional; also install the Tesseract .exe

# Cyrillic-script OCR for PHOTOGRAPHED documents (not screenshots) — see
# pii_blur.ocr._cyrillic_scan_pass. winocr's "ru" language pack already reads
# Cyrillic for normal screenshots; this is specifically for the degraded-photo
# path (pii_blur.scan_preprocess), where winocr/rapidocr-onnxruntime's default
# Chinese+English model only approximate Cyrillic glyphs with similar-looking
# Latin letters. The newer `rapidocr` package (successor to
# rapidocr-onnxruntime) auto-downloads a dedicated, Apache-2.0-licensed
# Cyrillic PP-OCRv5 model (~8MB, downloaded once on first use, not bundled in
# the installer) that reads Cyrillic natively instead of guessing shapes.
rapidocr>=3.0

# Neural name model (hybrid): a 64MB int8 ONNX BERT, run with onnxruntime (already
# pulled in by rapidocr) + the lightweight tokenizers lib (Rust-backed, no heavy
# transformers runtime needed).
tokenizers>=0.20

# Face blurring — uses OpenCV's bundled Haar cascade. opencv-python is already a
# transitive dependency of rapidocr-onnxruntime; pinned explicitly here since
# pii_blur.faces imports cv2 directly. Capped below 5.0: OpenCV 5's Python
# bindings dropped/renamed cv2.CascadeClassifier (confirmed by testing against
# opencv-python 5.0.0 — AttributeError), so "<5" keeps the well-established,
# long-stable 4.x face-detection API.
opencv-python>=4.8,<5

# Tray app
pystray>=0.19

pytest>=8.0
